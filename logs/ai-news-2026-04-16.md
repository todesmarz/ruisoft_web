(function() {
  const nav = document.getElementById('news-tab-nav');
  const buttons = Array.from(nav.querySelectorAll('.tab-btn'));
  const contents = Array.from(document.querySelectorAll('.tab-content'));

  function showTab(tabName) {
    buttons.forEach(b => b.classList.toggle('active', b.dataset.tab === tabName));
    contents.forEach(c => c.classList.toggle('active', c.id === `tab-${tabName}`));
    document.dispatchEvent(new CustomEvent('dailyNews:tabChanged', { detail: { tab: tabName } }));
  }

  nav.addEventListener('click', (ev) => {
    const btn = ev.target.closest('.tab-btn');
    if (!btn) return;
    showTab(btn.dataset.tab);
  });

  document.dispatchEvent(new CustomEvent('dailyNews:initialized', { detail: { date: '2026-04-16' } }));
})();
</script>
