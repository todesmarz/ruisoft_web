    const btn=event.target.closest('.news-tab-btn');
    if(!btn) return;
    const tab=btn.getAttribute('data-tab');
    root.querySelectorAll('.news-tab-btn').forEach(function(node){node.classList.toggle('active',node===btn);});
    document.querySelectorAll('[id^="tab-2026-05-07-"]').forEach(function(panel){panel.classList.toggle('active',panel.id==='tab-2026-05-07-'+tab);});
    document.dispatchEvent(new CustomEvent('daily-news:tab-changed',{detail:{date:'2026-05-07',tab:tab}}));
  });
})();
</script>
