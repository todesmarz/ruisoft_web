    (function () {
      const root = document.getElementById('ai-news-tabs-2026-05-02');
      if (!root) return;
      const btns = root.querySelectorAll('.tab-btn');
      const panels = root.querySelectorAll('.tab-panel');
      function activate(name){
        btns.forEach(b=>b.classList.toggle('active', b.dataset.tab===name));
        panels.forEach(p=>p.classList.toggle('active', p.id===`tab-${name}`));
        root.dispatchEvent(new CustomEvent('ai-news:tab-changed',{detail:{tab:name}}));
      }
      btns.forEach(btn=>btn.addEventListener('click',()=>activate(btn.dataset.tab)));
      root.dispatchEvent(new CustomEvent('ai-news:loaded',{detail:{date:'2026-05-02'}}));
    })();
