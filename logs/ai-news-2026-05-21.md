(function () {
  const wrap = document.querySelector('.ai-news-wrap');
  if (!wrap) return;
  const buttons = wrap.querySelectorAll('.ai-news-tab-btn');
  const panels = wrap.querySelectorAll('.ai-news-panel');
  wrap.dispatchEvent(new CustomEvent('ai-news:tabs:ready', { detail: { date: '2026-05-21' } }));
  buttons.forEach((btn) => {
    btn.addEventListener('click', () => {
      const tab = btn.getAttribute('data-tab');
      buttons.forEach(b => b.classList.toggle('active', b === btn));
      panels.forEach(p => p.classList.toggle('active', p.id === `tab-${tab}`));
      wrap.dispatchEvent(new CustomEvent('ai-news:tab:change', { detail: { tab } }));
    });
  });
})();
